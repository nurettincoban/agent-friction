Hooks
Shell commands that run automatically before or after Claude's tool calls, and can block bad edits.
Why for you: Your two recurring mistakes (editing the generated `schema_gen.py` and using deprecated legacy payment calls) can both be caught mechanically. A hook enforces the rule even when Claude overlooks CLAUDE.md, and exit code 2 sends the error message back to Claude so it can fix the edit itself.
// .claude/settings.json
{
  "hooks": {
    "PreToolUse": [
      {
        "matcher": "Edit|Write",
        "hooks": [{"type": "command", "command": "jq -r '.tool_input.file_path' | grep -q 'schema_gen.py' && { echo 'schema_gen.py is generated: edit api/openapi.yaml and run make codegen' >&2; exit 2; } || exit 0"}]
      }
    ],
    "PostToolUse": [
      {
        "matcher": "Edit|Write",
        "hooks": [{"type": "command", "command": ".claude/hooks/check_payments.sh"}]
      }
    ]
  }
}

# .claude/hooks/check_payments.sh  (chmod +x)
#!/bin/sh
f=$(jq -r '.tool_input.file_path')
case "$f" in *payments/legacy.py) exit 0;; esac
if grep -nE 'legacy_charge|payments\.legacy' "$f" >&2; then
  echo 'Deprecated payment API used: switch to PaymentClient' >&2
  exit 2
fi
exit 0

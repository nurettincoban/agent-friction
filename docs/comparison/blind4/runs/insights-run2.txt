{
  "hooks": {
    "PreToolUse": [
      {"matcher": "Edit|Write", "hooks": [{"type": "command", "command": ".claude/hooks/protect-migrations.sh"}]}
    ],
    "PostToolUse": [
      {"matcher": "Edit|Write", "hooks": [{"type": "command", "command": "cargo clippy -p kiln-core --quiet -- -D clippy::unwrap_used -D clippy::expect_used 2>&1 | tail -20"}]}

  }
}

# .claude/hooks/protect-migrations.sh
#!/bin/sh
f=$(jq -r '.tool_input.file_path')
case "$f" in
  */migrations/*) if git ls-files --error-unmatch "$f" >/dev/null 2>&1; then echo "Shipped migration $f is immutable; add a new migration instead." >&2; exit 2; fi;;
esac
exit 0

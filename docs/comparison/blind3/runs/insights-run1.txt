Hooks
Shell commands that run automatically before or after Claude uses a tool, and can block the action or send feedback back to Claude.
Why for you: Your two most repeated corrections were `console.*` instead of the structured logger and editing an already-deployed migration. Both are mechanical rules a hook can enforce on every edit, so you don't have to catch them yourself.
// .claude/settings.json
{
  "hooks": {
    "PreToolUse": [
      {
        "matcher": "Edit|Write",
        "hooks": [{ "type": "command", "command": ".claude/hooks/protect-migrations.sh" }]
      }
    ],
    "PostToolUse": [
      {
        "matcher": "Edit|Write",
        "hooks": [{ "type": "command", "command": ".claude/hooks/no-console.sh" }]
      }
    ]
  }
}

# .claude/hooks/protect-migrations.sh
#!/usr/bin/env bash
f=$(jq -r '.tool_input.file_path // empty')
if [[ "$f" == */prisma/migrations/* && -f "$f" ]]; then
  echo "Blocked: $f is an existing (deployed) migration. Create a new one with 'npx prisma migrate dev --name ...'." >&2
  exit 2
fi
exit 0

# .claude/hooks/no-console.sh
#!/usr/bin/env bash
f=$(jq -r '.tool_input.file_path // empty')
[[ "$f" == *.ts && "$f" != *.test.ts ]] || exit 0
if hits=$(grep -n 'console\.' "$f"); then
  echo "console.* found in $f. Use req.log or the shared structured logger:" >&2
  echo "$hits" >&2
  exit 2
fi
exit 0

# then: chmod +x .claude/hooks/*.sh
